cmake_minimum_required(VERSION 3.20)
project(NetworkLib LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(NETLIB_BUILD_UBUNTU_DEMOS "Build Linux raw socket/epoll demos" ON)
option(NETLIB_BUILD_WINDOWS_IOCP_DEMOS "Build Windows IOCP demos" ON)
option(NETLIB_BUILD_BOOST_ASIO_DEMOS "Build Boost.Asio demos" ON)
option(NETLIB_BUILD_GRPC_DEMOS "Build gRPC demos" OFF)
option(NETLIB_BUILD_COROUTINE_DEMOS "Build C++20 coroutine demos" ON)
option(NETLIB_BUILD_ASIO_GRPC_DEMOS "Build asio-grpc demos" OFF)

if(UNIX AND NETLIB_BUILD_UBUNTU_DEMOS)
  add_subdirectory(apps/ubuntu)
endif()

if(WIN32 AND NETLIB_BUILD_WINDOWS_IOCP_DEMOS)
  add_subdirectory(apps/windows)
endif()

if(NETLIB_BUILD_BOOST_ASIO_DEMOS)
  find_package(Boost 1.74 QUIET COMPONENTS system)
  if(Boost_FOUND)
    add_subdirectory(apps/asio)
  else()
    message(WARNING "Boost(system) not found, skip Boost.Asio demos")
  endif()
endif()

if(NETLIB_BUILD_GRPC_DEMOS)
  find_package(Protobuf CONFIG QUIET)
  find_package(gRPC CONFIG QUIET)
  if(Protobuf_FOUND AND gRPC_FOUND)
    add_subdirectory(apps/grpc)
  else()
    message(WARNING "gRPC/Protobuf not found, skip gRPC demos")
  endif()
endif()

if(NETLIB_BUILD_COROUTINE_DEMOS)
  find_package(Boost 1.74 QUIET COMPONENTS system)
  if(Boost_FOUND)
    add_subdirectory(apps/coroutine)
  else()
    message(WARNING "Boost(system) not found, skip coroutine demos")
  endif()
endif()


if(NETLIB_BUILD_ASIO_GRPC_DEMOS)
  find_package(Boost 1.74 QUIET COMPONENTS system)
  find_package(Protobuf CONFIG QUIET)
  find_package(gRPC CONFIG QUIET)
  if(Boost_FOUND AND Protobuf_FOUND AND gRPC_FOUND)
    add_subdirectory(apps/asio-grpc)
  else()
    message(WARNING "Boost(system)/gRPC/Protobuf not found, skip asio-grpc demos")
  endif()
endif()
